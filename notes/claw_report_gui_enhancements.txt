class ObservationsView(QTableWidget):
    def __init__(self, max_rows=500):
        super().__init__()
        self.setColumnCount(4)  # id | zulu | source | ok
        self.setHorizontalHeader(...)
        self.horizontalHeader().setSectionResizeMode(QHeaderView.Interactive)
        self.setSortingEnabled(True)
        
    def populate(self, cursor='asc'):
        con = db.connect()
        query = "SELECT * FROM observations ORDER BY id DESC LIMIT 500"
        if cursor == 'desc':
            query += " LIMIT 100"
        rows = con.execute(query).fetchall()
        
        for row_data in reversed(rows):  # Newest first
            row = self.rowCount()
            self.insertRow(row)
            self.setItem(row, 0, QTableWidgetItem(str(row_data['id'])))
            self.setItem(row, 1, QTableWidgetItem(row_data['zulu']))
            # Color-code source by domain (qt.io = cyan, textualize.io = magenta)
            src = row_data.get('source', '')
            bg_color = QColor('cyan' if 'qt.io' in src else 'magenta' if 'textual' in src else 'white')
            cell = QTableWidgetItem(row_data['source'])
            cell.setForeground(QBrush(bg_color))
            self.setItem(row, 2, cell)
            status_badge = row_data.get('ok', 0)
            badge = "✓" if status_badge else "✗"
            style = {1: 'normal', 0: 'red'}[status_badge]
            self.setItem(row, 3, QTableWidgetItem(badge))

B. KNOWLEDGE GRAPH (FACTS)

class KnowledgeGraph(QGraphicsView):
    def __init__(self):
        super().__init__()
        self.scene = QGraphicsScene()
        self._graph = GraphModel()
        
    def populate_facts(self, limit=200):
        con = db.connect()
        rows = con.execute(f"SELECT * FROM facts ORDER BY id DESC LIMIT {limit}").fetchall()
        
        for row in reversed(rows):
            fact = dict(row)
            node = GraphNode(
                title=fact['title'][:50],
                body=fact['body'][:120] if fact.get('body') else '',
                whi=fact['whi']
            )
            self._graph.add_node(node)
        
        self.setScene(self._graph.scene())

class BiteViewer(QSplitter):

    def play_bite(self, bite_data):
        path = Path(bite_data['path'])
        if path.suffix.lower() == '.wav':
            pixmap = AudioThumbnail(path)  # Generate waveform thumbnail
            self.thumbnail_label.setPixmap(pixmap)
            QTimer.singleShot(5000, lambda: self._play_audio(str(path)))

class EnhancedPalaceView(QTableView):
    """Improved Palace table viewer with sorting and filtering."""
    
    def __init__(self, parent=None):
        super().__init__(parent)
        
        # Initialize columns from schema
        self.init_columns()
        
        # Set model properties  
        self.setSelectionBehavior(QAbstractItemView.SelectRows)
        self.setSelectionMode(QAbstractItemView.ExtendedSelection)
        self.setAlternatingRowColors(True)
        self.setStyleSheet("""
            QTableView {
                background: qlineargradient(x1:0, y1:0, 
                                            x2:1, y2:1, 
                                            stop:0 #1a1a2e, 
                                            stop:1 #16213e);
                border: 2px solid #00d9ff;
                gridline-color: rgba(0, 217, 255, 0.2);
            }
            QTableView::item {
                padding: 4px;
            }
        """)
        
    def init_columns(self):
        """Populate columns from current table selection."""
        con = db.connect()
        con.execute(f"PRAGMA table_info({self.table_name})")
        cols = [row[1] for row in con.fetchall()]  # Column names
        
        for col_name in cols:
            self.setColumnCount(self.columnCount() + 1)
            header = QTableWidgetItem(col_name[:20])
            header.setFlags(header.flags() & ~Qt.ItemIsEditable)
            self.setHorizontalHeaderItem(self.columnCount() - 1, header)
            
    def set_table(self, table_name):
        """Switch to displaying a different table."""
        self.table_name = table_name
        
        con = db.connect()
        query = f"SELECT * FROM {table_name} LIMIT 300"
        rows = con.execute(query).fetchall()
        
        model = QtSqlModel(con.get_connection(), self.connection())
        model.set_select("SELECT * FROM %s" % table_name)
        self.setModel(model)
        
        # Customize column formats
        col_map = {
            'id': str,
            'zulu': lambda x: datetime.fromisoformat(x).strftime('%Y-%m-%d %H:%M:%S'),
            'title': str,
            'body': lambda x: (x[:80] + "...") if x and len(x) > 80 else x,
        }
        
        for col_idx, col_name in enumerate(cols):
            fmt_fn = col_map.get(col_name, str)
            delegate = QStyledItemDelegate()
            delegate.setInitialAlign(Qt.AlignLeft | Qt.AlignVCenter)
